## タスク概要
error-handling.md §5.1 / §6.4 で確定済のチャネル別 composable 4 本を本単位で初実装する。`useErrorMessage` はエラー識別子 → i18n 文言への変換 + 未マッピング時の Sentry fallthrough を担う中核。残り 3 本 (`useFormErrors` / `useNoticeErrors` / `useToastErrors`) は `useErrorMessage.errorToMessage` をラップし、それぞれ `<UFormField>` inline / `<UAlert>` / `useToast` の各チャネルに文言を流す薄いラッパである。戻り値はすべて interfaces.ts §4 (`ErrorMessageApi` / `FormErrorsApi` / `NoticeErrorsApi` / `ToastErrorsApi`) に従う。本タスクは Phase 2 の全 domain composable (TASK-0008〜0013) がエラーチャネルとして消費する基盤となる。

## 実装詳細
### 1. useErrorMessage 🔵
**信頼性**: 🔵 *error-handling.md §5.1 (実装コードそのまま) + interfaces.ts ErrorMessageApi*
`app/types/error-codes.ts` の `APP_ERROR_CODES` / `PG_ERROR_CODES` / `isAppError` / `isPgError` を使い、識別子 → 文言を変換する。`useI18n()` の `t` / `te` を使用。App 識別子は 1:1 で `t('errors.{code}')` を返す (context 不要)。PG SQLSTATE は `tWithContext(base, ctx)` で `errors.{base}.{ctx}` を引き、`te()` で存在しなければ `.generic` にフォールバックする。いずれにも該当しない場合は `Sentry.captureException(error, { tags: { reason: 'unmapped_error_code' } })` を呼んで `t('errors.generic')` を返す。App 識別子の分岐には TASK-0003 で `error-codes.ts` に追加済の `ALREADY_IN_GROUP` を含めること (architecture.md §既存 API マッピング 注3、interfaces.ts §1)。
**実装ファイル**: `app/composables/useErrorMessage.ts`
```ts
// app/composables/useErrorMessage.ts (error-handling.md §5.1)
import * as Sentry from '@sentry/nuxt'
import {
  APP_ERROR_CODES,
  PG_ERROR_CODES,
  isAppError,
  isPgError,
  type ErrorContext
} from '~/types/error-codes'

export function useErrorMessage() {
  const { t, te } = useI18n()

  function tWithContext(base: string, ctx: ErrorContext): string {
    const ctxKey = `${base}.${ctx}`
    return t(te(ctxKey) ? ctxKey : `${base}.generic`)
  }

  function errorToMessage(error: unknown, pgContext: ErrorContext = 'generic'): string {
    if (isAppError(error, APP_ERROR_CODES.NOT_AUTHENTICATED)) return t('errors.not_authenticated')
    if (isAppError(error, APP_ERROR_CODES.NOT_A_MEMBER)) return t('errors.not_a_member')
    if (isAppError(error, APP_ERROR_CODES.INVALID_GROUP_NAME)) return t('errors.invalid_group_name')
    if (isAppError(error, APP_ERROR_CODES.ALREADY_IN_GROUP)) return t('errors.already_in_group')
    if (isAppError(error, APP_ERROR_CODES.INVITATION_NOT_FOUND_BY_LINK)) return t('errors.invitation_not_found_by_link')
    if (isAppError(error, APP_ERROR_CODES.INVITATION_EXPIRED)) return t('errors.invitation_expired')
    if (isAppError(error, APP_ERROR_CODES.INVITATION_CODE_COLLISION_AFTER_RETRY)) return t('errors.invitation_code_collision_after_retry')

    if (isPgError(error, PG_ERROR_CODES.UNIQUE_VIOLATION)) return tWithContext('errors.unique_violation', pgContext)
    if (isPgError(error, PG_ERROR_CODES.FK_VIOLATION)) return tWithContext('errors.fk_violation', pgContext)
    if (isPgError(error, PG_ERROR_CODES.CHECK_VIOLATION)) return tWithContext('errors.check_violation', pgContext)
    if (isPgError(error, PG_ERROR_CODES.RLS_REJECTED)) return t('errors.rls_rejected')

    Sentry.captureException(error, { tags: { reason: 'unmapped_error_code' } })
    return t('errors.generic')
  }

  return { errorToMessage }
}
```

### 2. useFormErrors / useNoticeErrors / useToastErrors 🔵
**信頼性**: 🔵 *error-handling.md §6.4 (3 本の実装コードそのまま) + interfaces.ts §4*
3 本とも内部で `useErrorMessage().errorToMessage` を呼ぶ薄いラッパ。`useFormErrors` は `fieldErrors: Ref<Record<string, string>>` を保持し `setFieldError(field, error, ctx?)` / `clear(field?)` を提供。`useNoticeErrors` は `notice: Ref<string | null>` を保持し `setNotice(error, ctx?)` / `clear()` を提供。`useToastErrors` は `useToast()` を内部に持ち `showError(error, ctx?)` で `toast.add({ title, color: 'red' })` を呼ぶ。
**実装ファイル**: `app/composables/useFormErrors.ts` / `app/composables/useNoticeErrors.ts` / `app/composables/useToastErrors.ts`
```ts
// app/composables/useFormErrors.ts (error-handling.md §6.4)
import type { ErrorContext } from '~/types/error-codes'

export function useFormErrors() {
  const fieldErrors = ref<Record<string, string>>({})
  const { errorToMessage } = useErrorMessage()

  function setFieldError(field: string, error: unknown, pgContext?: ErrorContext) {
    fieldErrors.value[field] = errorToMessage(error, pgContext)
  }

  function clear(field?: string) {
    if (field) delete fieldErrors.value[field]
    else fieldErrors.value = {}
  }

  return { fieldErrors, setFieldError, clear }
}
```
`useNoticeErrors` (`notice` / `setNotice` / `clear`) と `useToastErrors` (`showError`) は error-handling.md §6.4 のコードをそのまま実装する。
