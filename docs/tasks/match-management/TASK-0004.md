## タスク概要

`matches` へ試合を insert する **Write composable** `app/composables/useCreateMatch.ts` を実装する。
`useCreatePlayer.ts` の **pending を try/finally で制御**するパターン（二重送信防止）を踏襲する。
`group_id` は `useCurrentGroup` から付与し、4 選手 + `video_source_type` / `video_source_url` +
`name` + `match_date` を insert する。同一カード（同 4 選手・同動画）でも成功する（REQ-104 /
EDGE-005、該当 UNIQUE は無い）。戻りは `ActionResult<MatchListItem>`。

入力検証（name 境界 / 4 選手相異 / 動画ソース）は **フォーム側（Phase 2）+ TASK-0002 の Zod の責務**で、
本 composable は insert 実行に専念する。RLS 拒否 / 複合 FK / distinct CHECK / 通信エラーは
`ActionResult.error` に詰めて返し、表示（toast）は page 側（ADR-007、EDGE-010）。

## 実装詳細

### 1. useCreateMatch composable 🔵
**信頼性**: 🔵 *useCreatePlayer.ts pending・try-finally / interfaces.ts UseCreateMatchReturn / REQ-002 / REQ-104 / matches_insert RLS*

`useCreatePlayer.ts` の Write パターン（pending → try で実行 → finally でリセット、group_id を
`useCurrentGroup` から付与）を踏襲。`CreateMatchInput`（camelCase）を `matches` の snake_case 列へ
マッピングして insert する。`.select(...).single()` で挿入行を取得し `MatchListItem` 形に返す
（teamA/teamB の名前は insert 戻りでは ID のみのため、表示更新は page 側の `useMatches().refresh()`
に委ねる。`ActionResult<MatchListItem>` の name 解決はベストエフォートでよい）。

**実装ファイル**: `app/composables/useCreateMatch.ts`

```typescript
/**
 * 【機能概要】: matches へ試合を insert する Write composable
 * 【実装方針】: useCreatePlayer.ts の pending/try-finally を踏襲。group_id は useCurrentGroup から付与。
 * 🔵 interfaces.ts UseCreateMatchReturn / REQ-002 / REQ-104 / matches_insert RLS
 */
import type { Database } from '~/types/supabase'
import type { CreateMatchInput, MatchListItem } from '~/types/match'
import { useCurrentGroup } from '~/composables/useCurrentGroup'

interface ActionResult<T> { data: T | null; error: unknown }

interface UseCreateMatchReturn {
  createMatch: (input: CreateMatchInput) => Promise<ActionResult<MatchListItem>>
  pending: Ref<boolean>
}

export function useCreateMatch(): UseCreateMatchReturn {
  const pending = ref(false)
  const client = useSupabaseClient<Database>()
  const currentGroup = useCurrentGroup()

  async function createMatch(input: CreateMatchInput): Promise<ActionResult<MatchListItem>> {
    pending.value = true
    try {
      // 【group_id 取得】: 未取得時は insert 不可。error を返し page 側 toast (EDGE-010) 🔵
      const gid = currentGroup.data.value?.group_id
      if (!gid) return { data: null, error: new Error('no_current_group') }

      // 【insert 実行】: camelCase → snake_case マッピング。同カードでも成功 (REQ-104/EDGE-005) 🔵
      const { data, error } = await client
        .from('matches')
        .insert({
          group_id: gid,
          name: input.name ?? null,
          match_date: input.matchDate,
          team_a_player1_id: input.teamAPlayer1Id,
          team_a_player2_id: input.teamAPlayer2Id,
          team_b_player1_id: input.teamBPlayer1Id,
          team_b_player2_id: input.teamBPlayer2Id,
          video_source_type: input.videoSourceType,
          video_source_url: input.videoSourceUrl
        })
        .select('id, name, match_date')
        .single()

      // 【エラー処理】: RLS 拒否 / 複合FK / distinct CHECK / 通信は ActionResult.error に詰める (EDGE-010) 🔵
      if (error) return { data: null, error }

      // 【成功戻り値】: 挿入行の最小射影を返す (選手名解決は page 側 refresh に委ねる) 🔵
      return {
        data: { id: data.id, name: data.name, matchDate: data.match_date } as unknown as MatchListItem,
        error: null
      }
    } finally {
      // 【pending リセット】: 成功・エラーを問わず必ず false に戻す (二重送信防止) 🔵
      pending.value = false
    }
  }

  return { createMatch, pending }
}
```

> 注: `ActionResult<MatchListItem>` の戻りで teamA/teamB の名前まで埋めるかは任意。表示の正は
> `useMatches().refresh()` 後の一覧（埋め込み解決済）であり、insert 戻りは成否分岐に使う（ADR-007 §補遺）。
> 実装時に `MatchListItem` の必須フィールドを満たす最小射影方針を 1 つに固定し記録する。
